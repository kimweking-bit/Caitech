0003|import Link from "next/link";
0004|import { useRouter, useSearchParams } from "next/navigation";
0005|import { FormEvent, Suspense, useEffect, useState } from "react";
0006|import { useAuth } from "@/components/auth/AuthProvider";
0007|import { AuthFormTitle, AuthShell } from "@/components/auth/AuthShell";
0008|import { Button } from "@/components/ui/Button";
0009|import { Input } from "@/components/ui/Input";
0010|import { PasswordInput } from "@/components/ui/PasswordInput";
0011|import { ApiError, fieldErrors } from "@/lib/api/client";
0012|import { portalHome } from "@/lib/portal/roles";
0013|import type { AuthUser } from "@/types/api";
0014|
0015|function safeNext(raw: string | null): string | null {
0016|  if (!raw) return null;
0017|  if (!raw.startsWith("/") || raw.startsWith("//")) return null;
0018|  return raw;
0019|}
0020|
0021|function LoginForm() {
0022|  const { login, isAuthenticated, user, loading: authLoading } = useAuth();
0023|  const router = useRouter();
0024|  const search = useSearchParams();
0025|  const next = safeNext(search.get("next"));
0026|
0027|  const [identifier, setIdentifier] = useState("");
0028|  const [password, setPassword] = useState("");
0029|  const [error, setError] = useState<string | null>(null);
0030|  const [fields, setFields] = useState<Record<string, string>>({});
0031|  const [submitting, setSubmitting] = useState(false);
0032|
0033|  function goAfterAuth(u: AuthUser | null | undefined) {
0034|    router.replace(next || portalHome(u));
0035|  }
0036|
0037|  useEffect(() => {
0038|    if (!authLoading && isAuthenticated) {
0039|      goAfterAuth(user);
0040|    }
0041|    // eslint-disable-next-line react-hooks/exhaustive-deps
0042|  }, [authLoading, isAuthenticated, user, next, router]);
0043|
0044|  async function onSubmit(e: FormEvent) {
0045|    e.preventDefault();
0046|    setError(null);
0047|    setFields({});
0048|    const id = identifier.trim();
0049|    if (!id || !password) {
0050|      setError("Enter your username or email and password.");
0051|      return;
0052|    }
0053|    setSubmitting(true);
0054|    try {
0055|      const me = await login(id, password);
0056|      goAfterAuth(me);
0057|    } catch (err) {
0058|      setFields(fieldErrors(err));
0059|      let msg =
0060|        err instanceof ApiError ? err.message : "Could not sign you in.";
0061|      if (
0062|        /no active account/i.test(msg) ||
0063|        /invalid credentials/i.test(msg) ||
0064|        /unable to log in/i.test(msg)
0065|      ) {
0066|        msg = "Those credentials do not match our records.";
0067|      }
0068|      setError(msg);
0069|    } finally {
0070|      setSubmitting(false);
0071|    }
0072|  }
0073|
0074|  return (
0075|    <AuthShell
0076|      variant="login"
0077|      eyebrow="Portal access"
0078|      headline="Build the skills behind what comes next."
0079|      lead="Sign in to your CAITECH learning, teaching, or operations workspace."
0080|      detail="Secure access · Nairobi CBD"
0081|    >
0082|      <AuthFormTitle
0083|        title="Sign in"
0084|        description="Use your CAITECH username or email."
0085|      />
0086|
0087|      <form onSubmit={onSubmit} className="space-y-5" noValidate>
0088|        {error ? (
0089|          <p
0090|            role="alert"
0091|            className="border border-line bg-ground px-3 py-2.5 text-sm text-petrol"
0092|          >
0093|            {error}
0094|          </p>
0095|        ) : null}
0096|
0097|        <Input
0098|          id="login-identifier"
0099|          label="Username or email"
0100|          name="username"
0101|          autoComplete="username"
0102|          value={identifier}
0103|          onChange={(e) => setIdentifier(e.target.value)}
0104|          required
0105|          disabled={submitting}
0106|          error={fields.username || fields.email}
0107|        />
0108|
0109|        <div className="space-y-2">
0110|          <PasswordInput
0111|            id="login-password"
0112|            label="Password"
0113|            name="password"
0114|            autoComplete="current-password"
0115|            value={password}
0116|            onChange={(e) => setPassword(e.target.value)}
0117|            required
0118|            disabled={submitting}
0119|            error={fields.password}
0120|          />
0121|          <div className="flex justify-end">
0122|            <Link
0123|              href="/forgot-password"
0124|              className="font-mono text-[11px] uppercase tracking-[0.1em] text-ink-muted underline-offset-4 hover:text-petrol hover:underline focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-petrol"
0125|            >
0126|              Forgot password?
0127|            </Link>
0128|          </div>
0129|        </div>
0130|
0131|        <Button
0132|          type="submit"
0133|          variant="lime"
0134|          className="w-full justify-center"
0135|          disabled={submitting || authLoading}
0136|        >
0137|          {submitting ? "Signing in…" : "Sign in"}
0138|        </Button>
0139|      </form>
0140|
0141|      <p className="mt-8 border-t border-line pt-6 text-sm text-ink-muted">
0142|        Don&apos;t have an account?{" "}
0143|        <Link
0144|          href={next ? `/register?next=${encodeURIComponent(next)}` : "/register"}
0145|          className="font-medium text-petrol underline-offset-4 hover:underline"
0146|        >
0147|          Create one
0148|        </Link>
0149|      </p>
0150|    </AuthShell>
0151|  );
0152|}
